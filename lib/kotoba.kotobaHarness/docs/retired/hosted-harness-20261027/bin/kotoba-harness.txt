#!/bin/sh
# usage: bin/kotoba-harness <project.edn> <validate|known|wrong|jev> [out-dir]
# Env: KOTOBA_BIN (default kotoba), KOTOBA_BROWSER_HOST (default ../amu/runtime/browser-host.mjs),
#      NBB (default nbb), OPENROUTER_API_KEY for jev. Optionally, on macOS, the key is read from the
#      Keychain item named by KOTOBA_HARNESS_KEYCHAIN_SERVICE / KOTOBA_HARNESS_KEYCHAIN_ACCOUNT.
set -eu
home=$(CDPATH= cd -- "$(dirname -- "$0")/.." && pwd)
export KOTOBA_HARNESS_HOME="$home"
export KOTOBA_BROWSER_HOST="${KOTOBA_BROWSER_HOST:-$home/../amu/runtime/browser-host.mjs}"
[ "${2:-}" = validate ] || [ -f "$KOTOBA_BROWSER_HOST" ] || { echo "KOTOBA_BROWSER_HOST not found: $KOTOBA_BROWSER_HOST" >&2; exit 78; }
if [ "${2:-}" = jev ] && [ -z "${OPENROUTER_API_KEY:-}" ] && [ -n "${KOTOBA_HARNESS_KEYCHAIN_SERVICE:-}" ]; then
  OPENROUTER_API_KEY=$(security find-generic-password -s "$KOTOBA_HARNESS_KEYCHAIN_SERVICE" \
    -a "${KOTOBA_HARNESS_KEYCHAIN_ACCOUNT:-OPENROUTER_API_KEY}" -w 2>/dev/null || true)
  export OPENROUTER_API_KEY
fi
# Run nbb from the harness root so a caller's own nbb.edn (and its git deps)
# is never picked up; make path arguments absolute first.
abs() { case "$1" in /*) printf '%s' "$1" ;; *) printf '%s/%s' "$PWD" "$1" ;; esac; }
project=$(abs "$1"); shift
method="${1:-}"; [ "$#" -gt 0 ] && shift
out=$(abs "${1:-runs}")
cd "$home"
exec "${NBB:-nbb}" -cp "$home/src" -m kotoba-harness.main "$project" "$method" "$out"
